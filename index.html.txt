    // Звуковой движок на Web Audio API
    function startPumpSound() {
      if (!state.soundOn) return;
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        pumpOsc = audioCtx.createOscillator();
        pumpGain = audioCtx.createGain();
        pumpOsc.type = 'sawtooth';
        pumpOsc.frequency.setValueAtTime(52, audioCtx.currentTime);
        pumpGain.gain.setValueAtTime(0.03, audioCtx.currentTime);
        pumpOsc.connect(pumpGain);
        pumpGain.connect(audioCtx.destination);
        pumpOsc.start();
      } catch (e) {}
    }

    function updatePumpSound(pressure) {
      if (!pumpOsc || !pumpGain || !state.soundOn) return;
      pumpOsc.frequency.setTargetAtTime(48 + pressure * 2.2, audioCtx.currentTime, 0.1);
      pumpGain.gain.setTargetAtTime(0.015 + (pressure / 12) * 0.035, audioCtx.currentTime, 0.1);
    }

    function stopPumpSound() {
      if (pumpOsc) {
        try { pumpOsc.stop(); pumpOsc.disconnect(); } catch (e) {}
        pumpOsc = null;
      }
    }
